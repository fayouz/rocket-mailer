Le widget est du JavaScript sans dépendance : il suffit de monter le composeur quand le conteneur existe, de le détruire au démontage et de relayer les changements de brouillon.

Chargez `embed.js` une fois, dans le `<head>` de votre application ou dynamiquement. Les exemples ci-dessous supposent que `window.RocketMailer` est disponible.

::code-group
```vue [RocketMailerComposer.vue]
<script setup lang="ts">
import { onBeforeUnmount, onMounted, ref, watch } from 'vue'

const props = defineProps<{
  draft?: { to?: string[], subject?: string, htmlBody?: string }
}>()
const emit = defineEmits<{ sent: [email: { id: string, subject: string }] }>()

const container = ref<HTMLElement>()
let mailer: { setDraft: (d: object) => void, destroy: () => void } | undefined
let ready = false

onMounted(() => {
  mailer = window.RocketMailer.mount(container.value!, {
    baseUrl: import.meta.env.VITE_ROCKET_MAILER_URL,
    applicationId: import.meta.env.VITE_ROCKET_MAILER_APP_ID,
    getToken: () => fetch('/rocket-mailer/token', { method: 'POST' })
      .then(r => r.json()).then(d => d.token),
    draft: props.draft,
    onReady: () => { ready = true },
    onSent: email => emit('sent', email),
  })
})

watch(() => props.draft, (draft) => {
  if (ready && draft) mailer?.setDraft(draft)
}, { deep: true })

onBeforeUnmount(() => mailer?.destroy())
</script>

<template>
  <div ref="container" />
</template>
```

```vue [Nuxt (app.vue / page)]
<script setup lang="ts">
// nuxt.config.ts :
// app: { head: { script: [{ src: 'https://mailer.exemple.com/embed.js', defer: true }] } }
const draft = ref({ to: ['client@exemple.com'], subject: 'Votre devis' })
</script>

<template>
  <!-- Le widget manipule le DOM : rendu côté client uniquement -->
  <ClientOnly>
    <RocketMailerComposer :draft="draft" @sent="email => console.log(email)" />
  </ClientOnly>
</template>
```

```tsx [RocketMailerComposer.tsx]
import { useEffect, useRef } from 'react'

type Draft = { to?: string[], subject?: string, htmlBody?: string }
type Mailer = { setDraft: (d: Draft) => void, destroy: () => void }

export function RocketMailerComposer({ draft, onSent }: {
  draft?: Draft
  onSent?: (email: { id: string, subject: string }) => void
}) {
  const container = useRef<HTMLDivElement>(null)
  const mailer = useRef<Mailer>()
  const ready = useRef(false)

  useEffect(() => {
    mailer.current = (window as any).RocketMailer.mount(container.current, {
      baseUrl: import.meta.env.VITE_ROCKET_MAILER_URL,
      applicationId: import.meta.env.VITE_ROCKET_MAILER_APP_ID,
      getToken: () => fetch('/rocket-mailer/token', { method: 'POST' })
        .then(r => r.json()).then(d => d.token),
      draft,
      onReady: () => { ready.current = true },
      onSent,
    })
    return () => mailer.current?.destroy()
    // Monté une seule fois : les brouillons suivants passent par setDraft.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [])

  useEffect(() => {
    if (ready.current && draft) mailer.current?.setDraft(draft)
  }, [draft])

  return <div ref={container} />
}
```
::

::note
L'`applicationId` et l'URL de Rocket Mailer ne sont pas des secrets : ils peuvent figurer dans la configuration du front. Seul le jeton `rma_…` doit rester sur le serveur.
::

## Typage (TypeScript)

```ts [rocket-mailer.d.ts]
interface RocketMailerDraft {
  /** "Name <email>": must be offered to the user or allowed for your application. */
  from?: string
  to?: string[]
  cc?: string[]
  bcc?: string[]
  subject?: string
  htmlBody?: string
  /** Template id or IRI: loaded in the composer when htmlBody is not given. */
  template?: string
  /** Values of the template's "{{ name }}" variables, flat or nested. */
  variables?: Record<string, unknown>
  /** Ids of attachments uploaded for this user by your backend. */
  attachments?: string[]
}

interface RocketMailerSentEmail {
  id: string
  subject: string
  to: string[]
  status: 'queued' | 'sent' | 'failed'
}

interface RocketMailerOptions {
  baseUrl: string
  applicationId: string
  getToken: () => string | Promise<string>
  draft?: RocketMailerDraft
  onReady?: () => void
  onSent?: (email: RocketMailerSentEmail) => void
  onError?: (error: unknown) => void
  title?: string
  minHeight?: number
  autoResize?: boolean
}

interface RocketMailerInstance {
  iframe: HTMLIFrameElement
  setDraft: (draft: RocketMailerDraft) => void
  destroy: () => void
}

interface Window {
  RocketMailer: {
    mount: (target: string | HTMLElement, options: RocketMailerOptions) => RocketMailerInstance
  }
}
```
